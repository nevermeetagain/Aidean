const { chromium } = require("/workspace/work/Aidean/node_modules/.pnpm/playwright@1.62.1/node_modules/playwright");
(async () => {
  const browser = await chromium.launch();
  const page = await browser.newPage({ viewport: { width: 1360, height: 900 } });
  page.setDefaultTimeout(100000);
  await page.goto("http://localhost:3000/admin/login");
  await page.locator('input[type="email"], input[name="email"]').first().fill("admin@aidean.com");
  await page.locator('input[type="password"]').first().fill("Admin123!@#");
  await page.locator('button[type="submit"]').first().click();
  await page.waitForURL((u) => !u.pathname.includes("/login"), { timeout: 100000 });
  await page.goto("http://localhost:3000/admin/prices", { waitUntil: "networkidle" });
  console.log("url after login:", page.url());
  await page.screenshot({ path: "/tmp/shots/admin-prices-before.png" });
  await page.getByRole("tab", { name: "限时优惠" }).click();
  await page.waitForTimeout(3000);
  await page.screenshot({ path: "/tmp/shots/admin-after-tab.png" });
  await page.waitForSelector("[data-testid=promo-panel] table, [data-testid=promo-empty]");
  await page.locator("[data-testid=promo-panel]").screenshot({ path: "/tmp/shots/admin-promos.png" });
  await page.getByRole("button", { name: "新建优惠" }).first().click();
  await page.waitForTimeout(500);
  await page.locator('[role=dialog]').first().screenshot({ path: "/tmp/shots/admin-dialog.png" }).catch(async () => page.screenshot({ path: "/tmp/shots/admin-dialog.png" }));
  // 提交一个与既有优惠重叠的窗口 → 期望内联错误
  await page.locator('[data-testid=promo-model-search]').fill("claude-3-5-haiku");
  await page.waitForTimeout(1500);
  const sel = page.locator('[role=dialog] select').first();
  const opts = await sel.locator("option").allTextContents();
  console.log("search options:", opts.slice(0, 4).join(" | "));
  const target = opts.find((t) => t.includes("Claude 3.5 Haiku"));
  if (target) await sel.selectOption({ label: target });
  await page.locator('[role=dialog] input[inputmode=decimal]').nth(0).fill("0.3");
  await page.locator('[role=dialog] input[inputmode=decimal]').nth(1).fill("1.5");
  await page.getByRole("button", { name: "保存" }).click();
  await page.waitForSelector('[role=dialog] [data-testid=promo-error]');
  console.log("overlap error:", await page.locator('[role=dialog] [data-testid=promo-error]').innerText());
  await page.locator('[role=dialog]').first().screenshot({ path: "/tmp/shots/admin-dialog-error.png" }).catch(() => {});
  // 成功路径：改选一个无优惠的模型 → 保存 → promo-saved
  await page.locator('[data-testid=promo-model-search]').fill("gpt-4o");
  await page.waitForTimeout(1500);
  const opts2 = await sel.locator("option").allTextContents();
  const t2 = opts2.find((t) => !t.startsWith("—"));
  console.log("success target:", t2);
  if (t2) await sel.selectOption({ label: t2 });
  await page.getByRole("button", { name: "设为限时免费" }).click();
  await page.getByRole("button", { name: "保存" }).click();
  await page.waitForSelector("[data-testid=promo-saved]");
  console.log("saved:", await page.locator("[data-testid=promo-saved]").innerText());
  await page.waitForTimeout(800);
  await page.locator("[data-testid=promo-panel]").screenshot({ path: "/tmp/shots/admin-promos-after-save.png" });
  await browser.close();
})().catch(async (e) => { console.error(e.message.slice(0, 300)); process.exit(1); });
